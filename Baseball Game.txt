class Solution {
    public int calPoints(String[] operations) {
        ArrayList<Integer> list=new ArrayList<>();
        int sum=0;

        for(String op:operations){
            int n=list.size();
            if(op.equals("+")){
                int val=list.get(n-1)+list.get(n-2);
                list.add(val);
                sum+=val;
            }
            else if(op.equals("D")){
                int val=2*list.get(n-1);
                list.add(val);
                sum+=val;
            }
            else if(op.equals("C")){
                sum-=list.remove(n-1);
            }
            else{
                int val=Integer.parseInt(op);
                list.add(val);
                sum+=val;
            }
        }
        return sum;
    }
}
