class Solution {
    public List<List<Integer>> combinationSum(int[] candidates, int target) {
        List<List<Integer>> ans=new ArrayList<>();
        List<Integer> l=new ArrayList<>();
        Permutation(candidates,target,l,0,ans);
        return ans;
    }
     public static void Permutation(int []coin,int amount,List<Integer> l,int idx,List<List<Integer>> ans){
        if(amount==0){
            //System.out.println(l);
            ans.add(new ArrayList<>(l));
            return;
        }
        for(int i=idx;i<coin.length;i++){
            if(amount>=coin[i]){
                l.add(coin[i]);
                //amount-=coin[i];
                Permutation(coin,amount-coin[i],l,i,ans); //last wala hatega
                //amount+=coin[i];
                l.remove(l.size()-1);
            }
        }
    }
}
