class Solution {
    public List<List<String>> partition(String s) {
        List<String> l = new ArrayList<>();
        List<List<String>> l1 = new ArrayList<>();

        Partition(s, l, l1);
        return l1;
    }
    public static void Partition(String ques, List<String> l,
                                 List<List<String>> l1) {
        if (ques.length() == 0) {
            l1.add(new ArrayList<>(l));
            return;
        }

        for (int cut = 1; cut <= ques.length(); cut++) {
            String s = ques.substring(0, cut);

            if (isPan(s)) {
                l.add(s);
                Partition(ques.substring(cut), l, l1);
                l.remove(l.size() - 1);
            }
        }
    }
    public static boolean isPan(String s) {
        int i = 0;
        int j = s.length() - 1;

        while (i < j) {
            if (s.charAt(i) != s.charAt(j)) {
                return false;
            }
            i++;
            j--;
        }
        return true;
    }
}S
