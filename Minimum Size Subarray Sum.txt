class Solution {
    public int minSubArrayLen(int target, int[] nums) {
        int n=nums.length;
        int s=1;
        int e=n;
        int min=0;

        while(s<=e){
            int mid=s+(e-s)/2;
            if(window(mid,target,nums)){
                min=mid;
                e=mid-1;
            }
            else{
                s=mid+1;
            }
        }
        return min;
    }
    private boolean window(int k,int target,int nums[]){
        int sum=0;
        for(int i=0;i<nums.length;i++){
            sum+=nums[i];
            if(i>=k){
                sum-=nums[i-k];
            }
            if(i>=k-1 && sum>=target){
                return true;
            }
        }
        return false;
    }
}
