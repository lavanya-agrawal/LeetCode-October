class Solution {
    public List<Integer> lexicalOrder(int n) {
        List<Integer> result = new ArrayList<>();
        for(int i=1;i<=9;i++){
            Lex(n,i,result);
        }
        return result;
    }
    public static void Lex(int n,int cur,List<Integer> result){
        if(cur>n){
            return;
        }
        result.add(cur);
        for(int i=0;i<=9;i++){
            Lex(n,cur*10+i,result);
        }
    }
}
