class Solution {
    public int countDigitOne(int n) {
        long count=0;
        for(long i=1;i<=n;i*=10){
            long d=i*10;
            long h=n/d;
            long cur=(n/i)%10;
            long l=n%i;

            long c=h*i;
            if(cur==1){
                c+=(l+1);
            }
            else if(cur>1){
                c+=i;
            }
            count+=c;
        }
        return (int)count;
    }
}
