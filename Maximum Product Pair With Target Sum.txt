class Solution {
    public int[] maxProductPair(int[] nums, int target) {
        int p=0;
        int max=Integer.MIN_VALUE;
        int a11=-1,b11=-1;

        for(int i=0;i<nums.length;i++){
            for(int j=0;j<nums.length;j++){
                if(i!=j && nums[i]+nums[j]==target && nums[i]>nums[j]){
                    p=nums[i]*nums[j];
                    if(p>max){
                        max=p;
                        a11=i;
                        b11=j;
                    }
                }
            }
        }
        // if(a11==-1){
        //     return new int[]{-1,-1};
        // }
        return new int[]{a11,b11};
    }
}
