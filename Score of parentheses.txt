class Solution {
    public int scoreOfParentheses(String s) {
        int score=0;
        int max=0;
        for(int i=0;i<s.length();i++){
            char ch=s.charAt(i);
            if(ch=='('){
                max++;
            }
            else{
                max--;
                if(s.charAt(i-1)=='('){
                    int s1=1;
                    for(int j=0;j<max;j++){
                        s1=s1*2;
                    }
                    score+=s1;
                }
            }
        }
        return score;
    }
}
