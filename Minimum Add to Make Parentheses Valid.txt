class Solution {
    public int minAddToMakeValid(String s) {
        int max=0;
        int min=0;

        for(char c:s.toCharArray()){
            if(c=='(') min++;
            else{
                if(min>0){
                    min--;
                }
                else{
                    max++;
                }
            }
        }
        return max+min;
    }
}
