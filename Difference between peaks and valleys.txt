class Solution {
    public int peakValleyDifference(int[] nums) {

        int sum=0;
        int sum1=0;

        for(int i=1;i<nums.length-1;i++){
            if(i>=0 && nums[i]>nums[i-1] && nums[i]>nums[i+1]){
                sum+=nums[i];
            }
        }
        for(int i=1;i<nums.length-1;i++){
            if(i>=0 && nums[i]<nums[i-1] && nums[i]<nums[i+1]){
                sum1+=nums[i];
            }
        }
        return sum-sum1;
    }
}
